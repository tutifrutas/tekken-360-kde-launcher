set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network)
find_package(SDL3 REQUIRED CONFIG)
find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(ECMFindQmlModule)
ecm_find_qmlmodule(org.kde.kirigami REQUIRED)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_library(tekken_launcher_core STATIC
    settings.cpp settings.h game_session.cpp game_session.h)
target_link_libraries(tekken_launcher_core PUBLIC Qt6::Core)
target_include_directories(tekken_launcher_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
qt_add_library(tekken_displays STATIC display_catalog.cpp display_catalog.h)
target_link_libraries(tekken_displays PUBLIC Qt6::Gui SDL3::SDL3)
qt_add_library(tekken_cover_cache STATIC cover_cache.cpp cover_cache.h)
target_include_directories(tekken_cover_cache PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(tekken_cover_cache PUBLIC Qt6::Core Qt6::Gui Qt6::Network)
qt_add_executable(tekken_launcher main.cpp)
qt_add_qml_module(tekken_launcher URI Tekken.Launcher VERSION 1.0
    QML_FILES Main.qml PlayPage.qml CoverCard.qml SettingsPanel.qml)
target_link_libraries(tekken_launcher PRIVATE tekken_launcher_core tekken_displays tekken_cover_cache Qt6::Gui Qt6::Quick Qt6::QuickControls2)
include(GNUInstallDirs)
install(TARGETS tekken_launcher RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES org.tekken.Native.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
if(BUILD_TESTING)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
    qt_add_executable(tekken_cover_cache_tests ../tests/launcher/cover_cache_test.cpp)
    target_link_libraries(tekken_cover_cache_tests PRIVATE tekken_cover_cache Qt6::Test)
    add_test(NAME native_cover_cache COMMAND tekken_cover_cache_tests)
    set_tests_properties(native_cover_cache PROPERTIES TIMEOUT 30)
    qt_add_executable(tekken_launcher_tests ../tests/launcher/launcher_test.cpp)
    target_link_libraries(tekken_launcher_tests PRIVATE tekken_launcher_core Qt6::Test)
    add_test(NAME native_launcher COMMAND tekken_launcher_tests)
    set_tests_properties(native_launcher PROPERTIES TIMEOUT 30)
    qt_add_executable(tekken_launcher_ui_tests ../tests/launcher/ui_test.cpp)
    target_link_libraries(tekken_launcher_ui_tests PRIVATE tekken_launcher_core tekken_displays Qt6::Test Qt6::Quick Qt6::QuickControls2)
    add_test(NAME native_launcher_ui COMMAND tekken_launcher_ui_tests)
    set_tests_properties(native_launcher_ui PROPERTIES TIMEOUT 20 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_test(NAME native_launcher_qml COMMAND tekken_launcher --smoke-test)
    set_tests_properties(native_launcher_qml PROPERTIES TIMEOUT 10 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
